cmake_minimum_required(VERSION 3.20)
project(VAST2009Explorer VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(VAST_BUILD_GUI "Build the VTK interactive application" ON)
add_library(vast_core src/Data.cpp)
target_include_directories(vast_core PUBLIC src)
if(MSVC)
  target_compile_options(vast_core PRIVATE /W4)
else()
  target_compile_options(vast_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(vast-analyze src/analyze.cpp)
target_link_libraries(vast-analyze PRIVATE vast_core)
if(VAST_BUILD_GUI)
  find_package(VTK 9 REQUIRED COMPONENTS CommonCore CommonDataModel
    RenderingCore RenderingOpenGL2 RenderingFreeType InteractionStyle IOImage)
  add_executable(vast-explorer src/Explorer.cpp)
  target_link_libraries(vast-explorer PRIVATE vast_core ${VTK_LIBRARIES})
  vtk_module_autoinit(TARGETS vast-explorer MODULES ${VTK_LIBRARIES})
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(vast-tests tests/test_data.cpp)
  target_link_libraries(vast-tests PRIVATE vast_core)
  add_test(NAME data-analysis COMMAND vast-tests "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
